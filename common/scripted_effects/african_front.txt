african_front_refresh_country = {
    african_front_calculate_sum = yes
    african_front_update_frames = yes
    if = {
        limit = { african_front_is_axis = yes }
        if = {
            limit = { NOT = { has_dynamic_modifier = { modifier = african_gui_modifier_axis } } }
            add_dynamic_modifier = { modifier = african_gui_modifier_axis }
        }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = african_gui_modifier_axis } }
        remove_dynamic_modifier = { modifier = african_gui_modifier_axis }
    }
    if = {
        limit = { african_front_is_allies = yes }
        if = {
            limit = { NOT = { has_dynamic_modifier = { modifier = african_gui_modifier_allies } } }
            add_dynamic_modifier = { modifier = african_gui_modifier_allies }
        }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = african_gui_modifier_allies } }
        remove_dynamic_modifier = { modifier = african_gui_modifier_allies }
    }
}
