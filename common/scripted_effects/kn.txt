    clr_country_flag = kn_incoming

kn_reset_board = {
    add_to_variable = { kn_serial = 1 }

kn_start_watching = {
    if = {
        limit = {
            FROM = {
                exists = yes
                has_country_flag = kn_active
                check_variable = { kn_mark = 1 }
                NOT = { tag = ROOT }
                var:kn_opponent = { exists = yes NOT = { tag = ROOT } }
            }
        }
        set_variable = { kn_watch_x = FROM }
        set_variable = { kn_watch_o = FROM.kn_opponent }
        set_variable = { kn_watch_serial = FROM.kn_serial }
        set_country_flag = kn_watch_open
    }
}
kn_send_invitation = {
    if = {
        limit = { kn_free = yes kn_invitation_ready = yes FROM = { exists = yes kn_free = yes NOT = { tag = ROOT } } }
        set_country_flag = { flag = kn_invitation_cooldown value = 1 days = 90 }
        kn_reset_board = yes
        set_variable = { kn_opponent = FROM }
        set_variable = { kn_mark = 1 }
        set_country_flag = kn_busy
        set_country_flag = kn_outgoing
        set_country_flag = kn_open
        set_country_flag = { flag = kn_invitation_live value = 1 days = 7 }
        FROM = {
            kn_reset_board = yes
            set_variable = { kn_opponent = ROOT }
            set_variable = { kn_mark = 2 }
            set_country_flag = kn_busy
            set_country_flag = kn_incoming
            set_country_flag = kn_open
            set_country_flag = { flag = kn_invitation_live value = 1 days = 7 }
        }
    }
}
kn_accept = {
    if = {
        limit = {
            has_country_flag = kn_incoming
            has_country_flag = kn_invitation_live
            kn_link_valid = yes
            var:kn_opponent = { has_country_flag = kn_outgoing has_country_flag = kn_invitation_live }
        }
        clr_country_flag = kn_incoming
        clr_country_flag = kn_invitation_live
        set_country_flag = kn_active
        var:kn_opponent = {
            clr_country_flag = kn_outgoing
            clr_country_flag = kn_invitation_live
            set_country_flag = kn_active
            set_country_flag = kn_open
        }
    }
}

kn_after_move = {
    add_to_variable = { kn_moves = 1 }
    set_variable = { kn_turn = 3 }
    subtract_from_variable = { kn_turn = kn_mark }
    var:kn_opponent = {
        set_variable = { kn_moves = PREV.kn_moves }
        set_variable = { kn_turn = PREV.kn_turn }
    }
    if = {
        limit = { kn_has_line = yes }
        kn_end_1_2 = yes
    }
    else_if = {
        limit = { check_variable = { kn_moves = 9 } }
        kn_end_3_3 = yes
    }
}

kn_daily_tick = {
    if = {
        limit = { has_country_flag = kn_busy }
        if = {
            limit = { NOT = { kn_link_valid = yes } }
            set_variable = { kn_result = 7 }
            kn_unlock = yes
            set_country_flag = kn_open
        }
        else_if = {
            limit = { NOT = { has_country_flag = kn_active } NOT = { has_country_flag = kn_invitation_live } }
            kn_end_6_6 = yes
        }
        else_if = {
            limit = { is_ai = yes has_country_flag = kn_incoming }
            kn_accept = yes
        }
        else_if = {
            limit = { is_ai = yes kn_can_move = yes }
            kn_ai_move = yes
        }
    }
}
