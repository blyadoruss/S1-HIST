on_actions = {
    on_startup = {
        effect = { african_front_update_control_flags = yes }
    }
    on_state_control_changed = {
        effect = { african_front_update_control_flags = yes }
    }
    on_daily = {
        effect = {
            if = {
                limit = {
                    OR = {
                        tag = ENG
                        AND = { tag = ITA NOT = { exists = ENG } }
                        AND = { tag = GER NOT = { exists = ENG } NOT = { exists = ITA } }
                    }
                }
                african_front_update_control_flags = yes
            }
        }
    }
}
